% Chapter (numbered)
\titleformat{\chapter}[display]
  {\sffamily\bfseries\LARGE\color{ChapterColor}\filright}
  {}% label built in the body
  {0pt}
  {%
    \vspace*{-2.5em}%
    \noindent
    \colorbox{ChapterBackground}{%
      \parbox[b]{\linewidth-2\fboxsep}{%
        \filright
        % Respect secnumdepth without \m@ne or \c@secnumdepth:
        \ifnum\value{secnumdepth}>-1
          \ChapterHeadingName\ \thechapter\\
        \fi
        #1%
      }%
    }%
  }
\titlespacing*{\chapter}{0pt}{0pt}{\baselineskip}
% Spacing to mimic your glue (-2.5em already applied above)
\titlespacing*{\chapter}{0pt}{0pt}{\baselineskip}

% Chapter (unnumbered)
\titleformat{name=\chapter,numberless}[display]
  {\sffamily\bfseries\LARGE\color{ChapterColor}\filright}
  {}{0pt}
  {%
    \vspace*{-2.5em}%
    \noindent
    \colorbox{ChapterBackground}{%
      \parbox{\linewidth-2\fboxsep}{%
        \filright #1%
      }%
    }%
  }
\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{\baselineskip}

% Paragraph (run-in head, unnumbered)
% Keep hierarchy shallow by not numbering at this level:
% \setcounter{secnumdepth}{3} (or lower) in your preamble if needed.
\titleformat{\paragraph}[runin]
  {\normalsize\sffamily\color{SectionColor}%\figureversion{tabular}
  }{}% empty label = no printed number
  {0pt}
  {#1:\ }% run-in delimiter
\titlespacing*{\paragraph}
  {0pt}{0.1ex plus 0.1ex minus 0.1ex}{0.5em}
  
% Subparagraph (run-in, italic)
\titleformat{\subparagraph}[runin]
  {\small\sffamily\itshape\color{SectionColor}%\figureversion{tabular}
  }{}{0pt}{#1\enspace}

  {0pt}{0.1ex plus 0.1ex minus 0.1ex}{0.5em}
