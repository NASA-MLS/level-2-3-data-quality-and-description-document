% Chapter (numbered)
\titleformat{\chapter}[display]
  {\sffamily\bfseries\LARGE\color{ChapterColor}\filright}
  {}% label built in the body
  {0pt}
  {%
    \vspace*{-2.5em}%
    \noindent
    \colorbox{ChapterBackground}{%
      \parbox[b]{\linewidth-2\fboxsep}{%
        \filright
        % Respect secnumdepth without \m@ne or \c@secnumdepth:
        \ifnum\value{secnumdepth}>-1
          \ChapterHeadingName\ \thechapter\\
        \fi
        #1%
      }%
    }%
  }
\titlespacing*{\chapter}{0pt}{0pt}{\baselineskip}
% Spacing to mimic your glue (-2.5em already applied above)
