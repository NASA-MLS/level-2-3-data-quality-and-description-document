\newcommand{\DrawNavBarProduct}[2]{%
  % #1 is the tag for the product (e.g., H2O)
  % #2 is the name for it (e.g., \ch{H2O})
  \ifthenelse{\equal{#1}{\currentnav}}%
  {%
    \def\nbnudge{\rule{2mm}{0pt}}%
    \colorlet{NBColor}{SelectedProductTabTextColor}%
    \colorlet{NBBackground}{SelectedProductTabBackroundColor}}%
  {%
    \def\nbnudge{\rule{0pt}{0pt}}%
    \colorlet{NBColor}{ProductTabTextColor}%
    \ifoddrow
      \colorlet{NBBackground}{ProductTabBackgroundA}%
    \else
      \colorlet{NBBackground}{ProductTabBackgroundB}%
    \fi}%
  \ifoddrow\global\oddrowfalse\else\global\oddrowtrue\fi%
  \def\nbsecref{sec:Standard#1}%
  \def\nbscrref{sec:#1Screening}%
  \def\nbtabref{tab:#1Summary}%
  \nbnudge\hyperref[\nbsecref]{\begin{tikzpicture}[outer sep=0pt, inner sep=5pt]
    \node (NBProduct#1) [nbentry] {#2};
    \node (NBProductRulesLink#1) [nblinks, anchor=north east]
      at (NBProduct#1.north west) {\hyperref[\nbscrref]{S}};
    \node (NBProductTableLink#1) [nblinks, anchor=south east]
      at (NBProduct#1.north east) {\hyperref[\nbtabref]{T}};
    % \node [anchor=east, inner sep=0pt, outer sep=0pt] at (NBProduct#1.west) {\nbnudge};
  \end{tikzpicture}}}

\newcommand{\DrawNavBarExtra}[3]{%
  % #1 is the label to link to
  % #2 is the text
  % #3 is the tikz label
  \ifthenelse{%
   {\equal{\thepage}{\pageref{tab:ProductSummary}}
     \and
   \equal{#2}{Table}}%
   \or
   \equal{#2}{\currentnav}
 }%
  {\def\nbnudge{\rule{2mm}{0pt}}%
    \colorlet{NBColor}{SelectedChapterTabTextColor}%
    \colorlet{NBBackground}{SelectedChapterTabBackroundColor}}%
  {\def\nbnudge{\rule{0pt}{0pt}}%
    \colorlet{NBColor}{ChapterTabTextColor}%
    \ifoddrow
      \colorlet{NBBackground}{ChapterTabBackgroundA}%
    \else
      \colorlet{NBBackground}{ChapterTabBackgroundB}%
    \fi}%
  \ifoddrow\global\oddrowfalse\else\global\oddrowtrue\fi%
  \nbnudge\hyperref[#1]{\begin{tikzpicture}[outer sep=0pt, inner sep=5pt]
      \node (#3) [nbentry] {#2};
      % \node [anchor=east, inner sep=0pt, outer sep=0pt] at (#3.west) {\nbnudge};
    \end{tikzpicture}}}
